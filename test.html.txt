    const stepIndicator = document.getElementById('stepIndicator');
    const progressFill = document.getElementById('progressFill');

    // -------- 渲染函数 --------
    function render() {
        if (currentStep < QUESTIONS.length) {
            renderQuestion();
        } else {
            renderResult();
        }
        // 更新进度
        stepIndicator.textContent = `${Math.min(currentStep + 1, QUESTIONS.length)} / ${QUESTIONS.length}`;
        const pct = (currentStep / QUESTIONS.length) * 100;
        progressFill.style.width = pct + '%';
    }

    function renderQuestion() {
        const q = QUESTIONS[currentStep];
        let html = `<div class="question-area">
            <div class="q-text">
                <div class="tag">🎯 场景 ${currentStep + 1}</div>
                ${q.text}
            </div>
            <div class="options">`;
        
        q.options.forEach((opt, idx) => {
            const letter = String.fromCharCode(65 + idx); // A,B,C
            html += `<button class="opt-btn" onclick="selectOption(${idx})">
                <span class="letter">${letter}</span> ${opt.label}
            </button>`;
        });

        html += `</div></div>`;
        container.innerHTML = html;
    }

    // 选择选项
    window.selectOption = function(optIdx) {
        const q = QUESTIONS[currentStep];
        const selected = q.options[optIdx];
        // 累加分数
        scores[selected.deptId] += 1;
        userAnswers.push(selected);
        // 进入下一题
        currentStep++;
        render();
    };

    // -------- 结果渲染（含MBTI风格） --------
    function renderResult() {
        // 找出最高分部门 (如有并列，取第一个)
        let maxScore = -1;
        let bestDeptId = 0;
        scores.forEach((score, id) => {
            if (score > maxScore) {
                maxScore = score;
                bestDeptId = id;
            }
        });

        const dept = DEPARTMENTS[bestDeptId];
        // 计算匹配度（满分6题，显示百分比）
        const matchPercent = Math.round((maxScore / 6) * 100);

        // 生成有趣的副词描述
        let vibe = '';
        if (matchPercent >= 83) vibe = '🔮 天选之人！';
        else if (matchPercent >= 67) vibe = '✨ 完美契合！';
        else if (matchPercent >= 50) vibe = '🌟 潜力无限！';
        else vibe = '🧚 隐藏天赋！';

        let html = `
            <div class="result-box">
                <div class="result-emoji">${dept.emoji}</div>
                <div class="result-title">${dept.name}</div>
                <div class="result-mbti">${dept.mbti} · ${dept.label}</div>

                </div>
                <div class="result-tags">
                    ${dept.tags.map(tag => `<span>${tag}</span>`).join('')}

            </div>
        `;

        stepIndicator.textContent = '🎉 结果';
    }
